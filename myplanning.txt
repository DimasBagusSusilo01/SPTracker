CREATE TABLE solar_telemetry (
    time TIMESTAMPTZ NOT NULL,
    device_id VARCHAR(32) NOT NULL,
    array_id VARCHAR(32) NOT NULL,
    power_kw NUMERIC(6,3),
    voltage_v NUMERIC(6,2),
    current_a NUMERIC(6,2),
    irradiance_wm2 INTEGER,
    panel_azimuth NUMERIC(5,2),
    panel_elevation NUMERIC(5,2),
    sun_azimuth NUMERIC(5,2),
    sun_elevation NUMERIC(5,2),
    alignment_error NUMERIC(4,2),
    cell_temp_c NUMERIC(5,2),
    ambient_temp_c NUMERIC(5,2),
    inverter_efficiency NUMERIC(5,2),
    system_mode VARCHAR(16) DEFAULT 'auto'
);

CREATE TABLE daily_production_summary (
    date DATE PRIMARY KEY,
    array_id VARCHAR(32) NOT NULL,
    total_yield_kwh NUMERIC(8,2),
    peak_power_kw NUMERIC(6,3),
    co2_saved_kg NUMERIC(8,2),
    revenue_usd NUMERIC(8,2),
    operational_hours NUMERIC(4,2),
    health_score NUMERIC(5,2)
);
